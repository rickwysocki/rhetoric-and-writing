\begin{frame}[label={sec:orgfbd956a}]{What is a Genre?}
A name for a typical or repeated response to a situation.
\end{frame}
\begin{frame}[label={sec:org88bbd9a}]{Genre Conventions}
\begin{itemize}
\item Content
\item Design/structure
\item Rhetorical strategies
\item Organization of ideas
\item Etc.
\end{itemize}
\end{frame}
\begin{frame}[label={sec:orgdccc720}]{Examples of Genre Conventions}
Rhetorical analysis essays:
\begin{itemize}
\item Thesis statements
\item A text to analyze
\item Academic formatting expectations (MLA/APA)
\item Typical structures (intro/body/conclusion)
\end{itemize}
\end{frame}
\begin{frame}[label={sec:orgf64cc15}]{Examples of Genre Conventions}
An email:
\begin{itemize}
\item A clear subject line.
\item A salutation, body text.
\item A closing.
\end{itemize}
\end{frame}
\begin{frame}[label={sec:org1bfd42f}]{Examples of Genre Conventions}
An action film: 
\begin{itemize}
\item Chase sequences.
\item Fights.
\item Explosions.
\end{itemize}
\end{frame}
\begin{frame}[label={sec:orga5ca3a8}]{Examples of Genre Conventions}
Trap music:
\begin{itemize}
\item Complicated hihat patterns.
\item 808 sub basses.
\item Typical themes in lyrical content.
\end{itemize}
\end{frame}
\begin{frame}[label={sec:orgbbb6684}]{Why are Genre Conventions Helpful?}
Genres are useful by presenting information in a format that matches the expectations of audiences.
\end{frame}
\begin{frame}[label={sec:orgdf21445}]{Genres and their Conventions Support Rhetorical Purposes}
\begin{block}{Rhetorical Analysis}
\begin{center}
\begin{tabular}{|l|p{0.6\textwidth}|}
Convention & Purpose\\
\hline
Thesis statement & Identifying the intellectual argument.\\
Focal text & Showing engagement with existing knowledge.\\
MLA-style formatting & Demonstrating your knowledge of academic writing norms.\\
\end{tabular}
\end{center}
\end{block}
\end{frame}
\begin{frame}[label={sec:orga7b319a}]{Genres and their Conventions Support Rhetorical Purposes}
\begin{block}{Action Films}
\begin{center}
\begin{tabular}{|l|p{0.6\textwidth}|}
Conventions & Purpose\\
\hline
Chase sequences & Putting the audience in the place of a thrilling scene .\\
Fights & Showing culturally situated values of strength and athleticism.\\
Explosions & Audiences like when movie goes boom.\\
\end{tabular}
\end{center}
\end{block}
\end{frame}
\begin{frame}[label={sec:org06b3494}]{Genre Thinking: Writers/Composers}
\begin{itemize}
\item General organizational patterns.
\item Resources to begin a project.
\item Understanding how to meet audience expectations.
\item Rhetorical success.
\end{itemize}
\end{frame}
\begin{frame}[label={sec:org6f16de9}]{Genre Thinking: Audiences}
\begin{itemize}
\item Audience comprehension.
\item Audience efficiency (e.g., the organization of a formal report is organized so that readers can read different sections).
\item Rhetorical success.
\end{itemize}
\end{frame}
\begin{frame}[label={sec:org4454e0b}]{Genres are not Just Patterns}
They are a "bundle" of conventions and audience expectations that are adapted and broken when necessary.
\end{frame}
