\begin{frame}[label={sec:orge029520}]{The Four Primary Rhetorical Appeals}
\begin{itemize}
\item \alert{Ethos:} Appeals to an audience's sense of character. \emph{What are ways that you assess the character of a speaker or writer?}
\item \alert{Pathos:} Appeals to an audience's sense of feeling, passion, or emotion. \emph{What examples of pathos appeals can you think of?}
\item \alert{Logos}: Appeals to an audience's sense of logic/reasoning, such as facts, statistics, quotes, testimonials, etc. \emph{What sorts of logos appeals do you find most persuasive?}
\item \alert{Kairos}: Appeals to an audience's sense of time and place. \emph{Can you think of failures in terms of Kairos?}
\end{itemize}
\end{frame}

\begin{itemize}

\begin{frame}[label={sec:org617e2ae}]{Aristotelian Analysis}
\begin{enumerate}
\item Considering the context of a situation: audience, relevant facts, time-period, etc.
\item Examining the text and how it presents its argument, most specifically by\ldots{}
\item Identifying the use of the primary rhetorical strategies (ethos, pathos, logos, and kairos), that are used to make the argument more or less persuasive.
\item Making an argument about whether the text should be believed based on points 1-3.
\end{enumerate}
