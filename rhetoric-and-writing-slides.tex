\begin{frame}[label={sec:org3417461}]{Contrast}
\begin{itemize}
\item Put simply, difference
\item Allows you to direct the reader’s attention
\item Helps you to group similar items together
\end{itemize}
\end{frame}

\begin{frame}[label={sec:org34a91a8}]{Repetition}
\begin{itemize}
\item Reusing the same or similar elements throughout a document/design
\item Suggests unity, consistency, cohesiveness
\end{itemize}
\end{frame}

\begin{frame}[label={sec:org4e0ca48}]{Alignment}
\begin{itemize}
\item Think about a grid on which you are ordering and arranging the various elements
\item Creates a sense of “clean-ness” to the design
\end{itemize}
\end{frame}

\begin{frame}[label={sec:orgf8971c0}]{Proximity}
\begin{itemize}
\item Moving things together or apart to achieve a more organized design
\item Allows you to group related items together and separate unrelated items
\end{itemize}
\end{frame}
